function showView(viewName) {
    document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
    document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));

    event.currentTarget.classList.add('active');
    $('page-title').textContent = viewName.charAt(0).toUpperCase() + viewName.slice(1);

function journeyTracker(step, status) {
    const steps = ['1','2','3','4'];
    let html = '<div class="journey">';
    steps.forEach((s,i) => {
        const num = i+1;
        let cls = '';
        if (num < step) cls = 'done';
        else if (num === step && status !== 'Completed' && status !== 'Dropped' && status !== 'Referred') cls = 'active';
        else if (num === step && status === 'Completed') cls = 'done';
        html += `<div class="journey-step ${cls}">${num}</div>`;
        if (i < steps.length-1) html += `<div class="journey-line ${num < step ? 'done' : ''}"></div>`;
    });
    return html + '</div>';
}

/* ================= SEARCH & FILTER ================= */
function searchBeneficiaries(q) {
    const term = q.toLowerCase();
    const filtered = store.beneficiaries.filter(b => 
        b.firstName.toLowerCase().includes(term) || 
        b.lastName.toLowerCase().includes(term) || 
        b.id.toLowerCase().includes(term)
    );
    renderBeneficiaries(filtered);
}

function filterBeneficiaries(key, value) {
    if (!value) { renderBeneficiaries(); return; }
    const filtered = store.beneficiaries.filter(b => b.status === value);
    renderBeneficiaries(filtered);
}

function addPlatform() {
    const name = $('p-name').value;
    if (!name) return alert('Please enter a platform name');
    const newP = {
        id: randId('PLT'), name, region: $('p-region').value, city: $('p-city').value,
        address: $('p-address').value, coordinator: $('p-coord').value || 'Unassigned',
        syncStatus: 'online', lastSync: new Date().toISOString(), beneficiaries: 0, active: 0, completed: 0
    };
    store.platforms.push(newP);
    renderPlatforms(); renderCloudDrives(); populateSelects();
    closeModal('modal-platform');
    // clear form
    ['p-name','p-region','p-city','p-address','p-coord','p-onedrive'].forEach(id => $(id).value = '');
}

function addBeneficiary() {
    const fname = $('b-fname').value, lname = $('b-lname').value;
    if (!fname || !lname) return alert('Please enter first and last name');
    const platformId = $('b-platform').value || 'PLT-001';
    const platform = store.platforms.find(p => p.id === platformId);
    const newB = {
        id: randId('YTR'), firstName: fname, lastName: lname,
        gender: $('b-gender').value, age: 20, ageGroup: '18-24',
        region: platform?.region || '', city: platform?.city || '',
        education: 'High School', registrationDate: new Date().toISOString().split('T')[0],
        platformId, program: $('b-program').value, status: 'Registered',
        email: $('b-email').value || '', phone: $('b-phone').value || '', journeyStep: 1
    };
    store.beneficiaries.unshift(newB);
    if (platform) { platform.beneficiaries++; platform.active++; }
    renderBeneficiaries(); renderPlatforms();
    // Update KPIs
    $('kpi-total').textContent = (parseInt($('kpi-total').textContent.replace(/,/g,'')) + 1).toLocaleString();
    closeModal('modal-beneficiary');
    ['b-fname','b-lname','b-email','b-phone'].forEach(id => $(id).value = '');
}

function sendMessage() {
    const subj = $('m-subject').value, body = $('m-body').value;
    if (!subj || !body) return alert('Please fill subject and message');
    store.messages.unshift({
        id: store.messages.length + 1, subject: subj, body, recipientType: $('m-recipient').options[$('m-recipient').selectedIndex].text,
        sentDate: new Date().toISOString().split('T')[0], status: 'sent', readCount: 0, totalCount: 1247
    });
    renderMessages();
    closeModal('modal-message');
    $('m-subject').value = ''; $('m-body').value = '';
}
